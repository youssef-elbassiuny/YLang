// 1. Simple function
fn add(a, b) {
    return a + b;
}

// 2. Function with if/else
fn check(x) {
    if (x > 0) {
        return 1;
    }
    elif(x<0){
        return 0;
    }

    return 0;

}

// 3. Function with if and return after it
fn absolute(x) {
    if (x < 0) {
        return 0 - x;
    }

    return x;
}

// 4. Function with elif and else
fn classify(x) {
    if (x > 0) {
        return 1;
    }
    elif (x < 0) {
        return 2;
    }
    else {
        return 3;
    }
}

// 5. Function with nested if
fn nested(x) {
    if (x > 0) {
        if (x > 10) {
            return 2;
        }
        else {
            return 1;
        }
    }
    else {
        return 0;
    }
}

// 6. Function with while
fn countdown(x) {
    while (x > 0) {
        x = x - 1;
    }

    return x;
}

// 7. Function where all branches return
fn all(x) {
    if (x > 0) {
        return 10;
    }
    elif (x < 0) {
        return 20;
    }
    else {
        return 30;
    }
}

// 8. Main program
let a = add(5, 3);
let b = check(5);
let c = absolute(0 - 7);
let d = classify(0);
let e = nested(15);
let f = countdown(5);
let g = all(1);

return a + b + c + d + e + f + g;
